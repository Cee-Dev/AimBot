local function getTargetPart(character)
    if aimBody then
        local hrp = character:FindFirstChild("HumanoidRootPart")
        if hrp then return hrp end
    end
    local head = character:FindFirstChild("Head")
    if head then return head end
    return character:FindFirstChild("HumanoidRootPart")
end

local function getValidTargets()
    local targets = {}
    local myTeam  = plr.TeamColor
    local myHead  = plr.Character and plr.Character:FindFirstChild("Head")
    if not myHead then return targets end
    local origin = myHead.Position
    local center = Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2)

    for _, v in pairs(Players:GetPlayers()) do
        if v ~= plr and v.Character
            and v.Character:FindFirstChild("Humanoid")
            and v.Character.Humanoid.Health > 0 then
            if v.TeamColor == myTeam then continue end
            local targetPart = getTargetPart(v.Character)
            if targetPart then
                local screenPos, onScreen = cam:WorldToViewportPoint(targetPart.Position)
                if onScreen then
                    local distFromCenter = (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude
                    local dist3D = (targetPart.Position - origin).Magnitude
                    if distFromCenter <= fovRadius then
                        table.insert(targets, {
                            player     = v,
                            part       = targetPart,
                            screenDist = distFromCenter,
                            dist3D     = dist3D,
                        })
                    end
                end
            end
        end
    end
    return targets
end

local function setCharm(player, active)
    if not player or not player.Character then return end
    for _, part in pairs(player.Character:GetDescendants()) do
        if part:IsA("BasePart") and not part:FindFirstChildWhichIsA("Highlight") then
            if active then
                local hl = Instance.new("Highlight")
                hl.FillColor          = Color3.fromRGB(255, 0, 0)
                hl.OutlineColor       = Color3.fromRGB(255, 0, 0)
                hl.FillTransparency   = 0.5
                hl.OutlineTransparency = 0.2
                hl.Parent             = part
            end
        end
    end
end

local function clearCharm(player)
    if not player or not player.Character then return end
    for _, hl in pairs(player.Character:GetDescendants()) do
        if hl:IsA("Highlight") then hl:Destroy() end
    end
end

local function updateCharm(targets)
    local activePlayers = {}
    for _, t in ipairs(targets) do activePlayers[t.player] = true end
    for p in pairs(highlightedPlayers) do
        if not activePlayers[p] then
            clearCharm(p)
            highlightedPlayers[p] = nil
        end
    end
    for p in pairs(activePlayers) do
        if not highlightedPlayers[p] then
            setCharm(p, true)
            highlightedPlayers[p] = true
        end
    end
end

-- ── Toggle Switch ───────────────────────────
local function makeToggle(parent, label, default, order, callback)
    local row = Instance.new("Frame")
    row.Name            = "Tog_" .. label
    row.Size            = UDim2.new(1, 0, 0, 34)
    row.BackgroundColor3 = C.Surface
    row.BorderSizePixel = 0
    row.LayoutOrder     = order
    row.Parent          = parent
    applyCorner(row, 5)

    -- hover
    row.MouseEnter:Connect(function()
        TweenService:Create(row, TWEEN_FAST, {BackgroundColor3 = C.SurfaceHov}):Play()
    end)
    row.MouseLeave:Connect(function()
        TweenService:Create(row, TWEEN_FAST, {BackgroundColor3 = C.Surface}):Play()
    end)

    local lbl = Instance.new("TextLabel")
    lbl.Size               = UDim2.new(1, -60, 1, 0)
    lbl.Position           = UDim2.new(0, 14, 0, 0)
    lbl.BackgroundTransparency = 1
    lbl.Text               = label
    lbl.TextColor3         = C.Text
    lbl.TextSize           = 12
    lbl.Font               = Enum.Font.GothamSemibold
    lbl.TextXAlignment     = Enum.TextXAlignment.Left
    lbl.Parent             = row

    -- track
    local track = Instance.new("Frame")
    track.Size            = UDim2.new(0, 36, 0, 18)
    track.Position        = UDim2.new(1, -48, 0.5, -9)
    track.BackgroundColor3 = default and C.Accent or C.ToggleOff
    track.BorderSizePixel = 0
    track.Parent          = row
    applyCorner(track, 9)

    -- knob
    local knob = Instance.new("Frame")
    knob.Size            = UDim2.new(0, 14, 0, 14)
    knob.Position        = default and UDim2.new(1, -16, 0.5, -7) or UDim2.new(0, 2, 0.5, -7)
    knob.BackgroundColor3 = C.Text
    knob.BorderSizePixel = 0
    knob.Parent          = track
    applyCorner(knob, 7)

    local state = default

    local function setState(v, skipCb)
        state = v
        TweenService:Create(track, TWEEN_FAST, {
            BackgroundColor3 = state and C.Accent or C.ToggleOff
        }):Play()
        TweenService:Create(knob, TWEEN_FAST, {
            Position = state and UDim2.new(1, -16, 0.5, -7) or UDim2.new(0, 2, 0.5, -7)
        }):Play()
        if not skipCb and callback then callback(state) end
    end

    -- click area
    local btn = Instance.new("TextButton")
    btn.Size               = UDim2.new(1, 0, 1, 0)
    btn.BackgroundTransparency = 1
    btn.Text               = ""
    btn.Parent             = row
    btn.MouseButton1Click:Connect(function() setState(not state) end)

    local ref = {setState = setState, getState = function() return state end}
    toggleRefs[label] = ref
    return ref
end

    local row = Instance.new("Frame")

    row.Size            = UDim2.new(1, 0, 0, 50)
    row.BackgroundColor3 = C.Surface
    row.BorderSizePixel = 0
    row.LayoutOrder     = order
    row.Parent          = parent
    applyCorner(row, 5)

    local lbl = Instance.new("TextLabel")
    lbl.Size               = UDim2.new(0.6, 0, 0, 22)
    lbl.Position           = UDim2.new(0, 14, 0, 2)
    lbl.BackgroundTransparency = 1
    lbl.Text               = label
    lbl.TextColor3         = C.Text
    lbl.TextSize           = 12
    lbl.Font               = Enum.Font.GothamSemibold
    lbl.TextXAlignment     = Enum.TextXAlignment.Left
    lbl.Parent             = row

    local valLbl = Instance.new("TextLabel")
    valLbl.Name            = "Val"
    valLbl.Size            = UDim2.new(0.35, 0, 0, 22)
    valLbl.Position        = UDim2.new(0.65, -14, 0, 2)
    valLbl.BackgroundTransparency = 1
    valLbl.Text            = tostring(default)
    valLbl.TextColor3      = C.Accent
    valLbl.TextSize        = 12
    valLbl.Font            = Enum.Font.GothamBold
    valLbl.TextXAlignment  = Enum.TextXAlignment.Right
    valLbl.Parent          = row

    -- track bg
    local trackBg = Instance.new("Frame")
    trackBg.Size            = UDim2.new(1, -28, 0, 4)
    trackBg.Position        = UDim2.new(0, 14, 0, 34)
    trackBg.BackgroundColor3 = C.ToggleOff
    trackBg.BorderSizePixel = 0
    trackBg.Parent          = row
    applyCorner(trackBg, 2)

    -- fill
    local fill = Instance.new("Frame")
    fill.Size            = UDim2.new((default - min) / (max - min), 0, 1, 0)
    fill.BackgroundColor3 = C.Accent
    fill.BorderSizePixel = 0
    fill.Parent          = trackBg
    applyCorner(fill, 2)

    -- knob
    local sKnob = Instance.new("Frame")
    sKnob.Size            = UDim2.new(0, 12, 0, 12)
    sKnob.Position        = UDim2.new((default - min) / (max - min), -6, 0.5, -6)
    sKnob.BackgroundColor3 = C.Text
    sKnob.BorderSizePixel = 0
    sKnob.ZIndex          = 3
    sKnob.Parent          = trackBg
    applyCorner(sKnob, 6)

    local dragging = false

    local function update(inputX)
        local tPos  = trackBg.AbsolutePosition.X
        local tSize = trackBg.AbsoluteSize.X
        local rel   = math.clamp((inputX - tPos) / tSize, 0, 1)
        local val   = math.floor(min + rel * (max - min))
        fill.Size     = UDim2.new(rel, 0, 1, 0)
        sKnob.Position = UDim2.new(rel, -6, 0.5, -6)
        valLbl.Text   = tostring(val)
        if callback then callback(val) end
    end

    -- invisible interaction layer over the track area
    local hitArea = Instance.new("TextButton")
    hitArea.Size               = UDim2.new(1, -10, 0, 22)
    hitArea.Position           = UDim2.new(0, 5, 0, 24)
    hitArea.BackgroundTransparency = 1
    hitArea.Text               = ""
    hitArea.ZIndex             = 4
    hitArea.Parent             = row

    hitArea.MouseButton1Down:Connect(function(x)
        dragging = true
        update(x)
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
            update(input.Position.X)
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = false
        end
    end)

    return {
        setValue = function(v)
            local rel = math.clamp((v - min) / (max - min), 0, 1)
            fill.Size      = UDim2.new(rel, 0, 1, 0)
            sKnob.Position = UDim2.new(rel, -6, 0.5, -6)
            valLbl.Text    = tostring(v)
        end
    }
end

local function switchTab(name)
    currentTab = name
    for n, btn in pairs(tabBtns) do
        local active = (n == name)
        TweenService:Create(btn, TWEEN_FAST, {
            TextColor3 = active and C.Accent or C.TextMuted,
        }):Play()
        btn:FindFirstChild("Ind").Visible = active
        tabPages[n].Visible = active
    end
end

for n, btn in pairs(tabBtns) do
    btn.MouseButton1Click:Connect(function() switchTab(n) end)
end

-- ══════════════════════════════════════════════
--  SYNC HELPERS  (keybind → UI)
-- ══════════════════════════════════════════════
local function syncUI()
    -- Toggles (skipCallback = true to avoid re‑setting vars)
    if toggleRefs["Aim Lock"]          then toggleRefs["Aim Lock"].setState(aiming, true) end
    if toggleRefs["Aim at Body"]       then toggleRefs["Aim at Body"].setState(aimBody, true) end
    if toggleRefs["FOV Circle Visible"] then toggleRefs["FOV Circle Visible"].setState(fovEnabled, true) end
    if toggleRefs["Charm / Highlight"] then toggleRefs["Charm / Highlight"].setState(charmEnabled, true) end

    -- Info rows
    infoAimLock.Text  = aiming and "ON" or "OFF"
    infoAimPart.Text  = aimBody and "Body" or "Head"
    infoFov.Text      = fovEnabled and "ON" or "OFF"
    infoFovRad.Text   = tostring(fovRadius)
    infoCharm.Text    = charmEnabled and "ON" or "OFF"
end

-- ══════════════════════════════════════════════
--  INPUT HANDLING  (original keybinds + UI toggle)
-- ══════════════════════════════════════════════
UserInputService.InputBegan:Connect(function(input, gpe)
    -- [=] toggles UI regardless of gameProcessedEvent
    if input.KeyCode == Enum.KeyCode.Equals then
        uiVisible = not uiVisible
        if uiVisible then
            outer.Visible = true
            outer.Size = UDim2.new(0, 420, 0, 0)
            outer.Position = outer.Position -- keep current pos
            TweenService:Create(outer, TWEEN_MED, {
                Size = minimized and UDim2.new(0, 420, 0, 42) or UDim2.new(0, 420, 0, 360)
            }):Play()
        else
            TweenService:Create(outer, TWEEN_MED, {
                Size = UDim2.new(0, 420, 0, 0),
            }):Play()
            task.delay(0.3, function() outer.Visible = false end)
        end
        return
    end

    if gpe then return end

    if input.KeyCode == Enum.KeyCode.Q then
        aiming = true
        syncUI()
    end

    if input.KeyCode == Enum.KeyCode.Minus then
        fovEnabled = not fovEnabled
        fovCircle.Visible = fovEnabled
        syncUI()
    end

    if input.KeyCode == Enum.KeyCode.RightBracket then
        charmEnabled = not charmEnabled
        if not charmEnabled then
            for p in pairs(highlightedPlayers) do clearCharm(p) end
            highlightedPlayers = {}
        end
        syncUI()
    end

    if input.KeyCode == Enum.KeyCode.LeftBracket then
        aimBody = not aimBody
        syncUI()
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if input.KeyCode == Enum.KeyCode.Q then
        aiming = false
        syncUI()
    end
end)

-- ══════════════════════════════════════════════
--  RENDER LOOP  (original — untouched + status)

local targetCount = 0

RunService.RenderStepped:Connect(function()
    fovCircle.Position = Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2)

    local targets = getValidTargets()
    targetCount = #targets

    if aiming then
        local closest = nil
        local shortestScreenDist = math.huge
        for _, t in ipairs(targets) do
            if t.screenDist < shortestScreenDist then
                shortestScreenDist = t.screenDist
                closest = t
            end
        end
        if closest then
            cam.CFrame = CFrame.new(cam.CFrame.Position, closest.part.Position)
        end
    end

    if charmEnabled then
        updateCharm(targets)
    else
        for p in pairs(highlightedPlayers) do clearCharm(p) end
        highlightedPlayers = {}
    end
end)
